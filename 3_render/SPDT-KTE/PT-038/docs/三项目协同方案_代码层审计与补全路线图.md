### 2.3 最小补全方案（不改管线架构，只加字段）

**方案：不修改管线，只在 episode .py 增加元数据字段**

在每个 episode .py 的 `dramatic_beats` 数组中，为每个 beat 增加可选字段：

```python

    "knowledge_type": "II",              # I/II/III/IV，I=强类比适用，II=受限类比，III=结构推导，IV=事实性
    "emotion_trigger": "curiosity",    # 受蕴情绪：curiosity/resonance/solemnity/sober_reflection
    "analogy_block": {
        "applicable": True,

        "target_domain": "颜体书法的力量感",

        "difference": "刀痕是破坏性的，笔锋是建构性的"

    "five_skandha": {
        "color_base": "关键词：蒲州、墨色、焦渴",
        "discrimination": "本 beat 对应'颜体书法的力量感来源于军旅生涯'",
```

**工作量估算**：无需修改生成器，只需在人工撰写/AI 辅助撰写 episode .py 时增加字段——这是最小干预。

```python
# scene_quality_gate.py 新增

def gate_G06_cognitive_load(scenes_dir):
    """检查每个 scene 的认知负荷（工作记忆 4±1）"""
    issues = []
    for f in sorted(os.listdir(scenes_dir)):
        if not f.endswith('.json'): continue
        with open(os.path.join(scenes_dir, f), encoding='utf-8-sig') as fh:
            data = json.load(fh)
        # 统计 content 中独立信息点数量（关键词、标签、图例项等）
        points = _count_info_points(data.get('content', {}))
        if points > 4:
            issues.append(f'{f}: 认知负荷{points}点 > 4，建议拆分')
    return len(issues)==0, issues

def gate_G07_five_skandha_completeness(scenes_dir):
    """检查 five_skandha 字段完整性"""
    issues = []
    for f in sorted(os.listdir(scenes_dir)):
        if not f.endswith('.json'): continue
        with open(os.path.join(scenes_dir, f), encoding='utf-8-sig') as fh:
            data = json.load(fh)
        required = ['color_base', 'discrimination', 'action_prompt']
        missing = [k for k in required if not data.get('five_skandha', {}).get(k)]
        if missing:
            issues.append(f'{f}: five_skandha 缺字段 {missing}')
    return len(issues)==0, issues

def gate_G08_dual_coding_alignment(scenes_dir):
    """检查声尘-色尘语义对齐"""
    issues = []
    for f in sorted(os.listdir(scenes_dir)):
        if not f.endswith('.json'): continue
        with open(os.path.join(scenes_dir, f), encoding='utf-8-sig') as fh:
            data = json.load(fh)
        if data.get('content', {}).get('scene_type') in ('concept', 'contrast', 'steps'):
            if not data.get('semantic_alignment'):
                issues.append(f'{f}: 核心 scene 缺少 semantic_alignment 字段')
    return len(issues)==0, issues
```

### 4.4 最小补全方案

**4.4.1 实现视频播放接口（MediaService.ets 补全）**

```typescript
// MediaService.ets，VideoPlayerInterface 实现
export async function playVideo(mediaRef: MediaRef): Promise<void> {
  if (mediaRef.type !== MediaType.VIDEO) return;
  await this.stop();

  this.avPlayer = await media.createAVPlayer();
  const fileUrl: string = 'file://' + mediaRef.src;
  this.avPlayer.url = fileUrl;

  this.avPlayer.on('stateChange', (state: string) => {
    // 同音频状态机逻辑
    if (state === 'completed') { this._state.state = PlayState.STOPPED; }
  });

  this._state.currentMedia = mediaRef;
  this._state.state = PlayState.PLAYING;
  this.emitState();
  await this.avPlayer.play();
  this.startProgressTimer();
}
```

**4.4.2 增加视频包导入逻辑（ImportService.ets 新增）**

```typescript
/**
 * 导入 video_factory 输出的 video_package
 * 包含: metadata.json + ep*.mp4 + ep*_scenes.json
 */
export async function importVideoPackage(
  metadataJson: string,
  videoFiles: string[]
): Promise<number> {
  const meta = JSON.parse(metadataJson) as VideoPackageMeta;
  const now = Date.now();

  for (const ep of meta.episodes) {
    // 1. 复制 MP4 到沙箱
    const destDir = VIDEO_DIR + ep.video_file.replace('.mp4', '/');
    // 2. 解析 scenes.json，建立时间轴
    const timeline = ep.scene_timeline ?? [];
    // 3. 写入视频包索引 DB
    await db.upsertVideoPackage({
      ep_id: ep.ep,
      ep_title: ep.title,
      video_path: destDir + ep.video_file,
      duration: ep.duration_seconds,
      chains: ep.chains,
      timeline: JSON.stringify(timeline),
      created_at: now
    });
  }
  return meta.episodes.length;
}
```
